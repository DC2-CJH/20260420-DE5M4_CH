Table orders {
  id integer
  cust_id integer
  created_at timestamp
}

Table customers {
  id integer [primary key]
  forename varchar
  surname varchar
  valid_flag integer
}

Table products {
  id integer [primary key]
  product_type varchar
  cust_id integer [primary key]
  order_id varchar
  status varchar
  created_at timestamp
}

Table ordered_items {
  id integer [primary key]
  order_id varchar
  item varchar
  order_date timestamp
}
